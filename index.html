Sending to Google Sheets…

🔧 Connect to Google Sheets

One-time setup — takes 5 minutes. After this, every submission goes directly to your sheet.

Step 1 — Create the Google Sheet

Go to sheets.new → name it "Pulse Check Responses".

Step 2 — Open Apps Script

Click Extensions → Apps Script. Delete existing code and paste this:

const SHEET_NAME = "Responses"; function doPost(e) { try { const ss = SpreadsheetApp.getActiveSpreadsheet(); let sheet = ss.getSheetByName(SHEET_NAME); if (!sheet) { sheet = ss.insertSheet(SHEET_NAME); const headers = ["Timestamp","Name","Month", "Goal 1","G1 Status","Goal 2","G2 Status","Goal 3","G3 Status", "Goal 4","G4 Status","Goal 5","G5 Status","Section 1 Comment", "Blocker","Blocker Description", "Course 1","C1 Status","C1 Score","Course 2","C2 Status","C2 Score", "Course 3","C3 Status","C3 Score","Course 4","C4 Status","C4 Score", "Course 5","C5 Status","C5 Score", "Key Development Need","Micro-Win","Action Required"]; sheet.appendRow(headers); sheet.getRange(1,1,1,headers.length) .setFontWeight("bold").setBackground("#1A3A5C").setFontColor("#FFFFFF"); } const d = JSON.parse(e.postData.contents); sheet.appendRow([ new Date().toLocaleString("en-GB"), d.name, d.month, d.g1name,d.g1status, d.g2name,d.g2status, d.g3name,d.g3status, d.g4name,d.g4status, d.g5name,d.g5status, d.s1comment, d.blocker, d.blockerDesc, d.c1name,d.c1status,d.c1score, d.c2name,d.c2status,d.c2score, d.c3name,d.c3status,d.c3score, d.c4name,d.c4status,d.c4score, d.c5name,d.c5status,d.c5score, d.s4a, d.s4b, d.action ]); const lastRow = sheet.getLastRow(); const actionCell = sheet.getRange(lastRow, 34); if (d.action==="No Meeting") actionCell.setBackground("#C6EFCE").setFontColor("#276221"); if (d.action==="15-Min Call") actionCell.setBackground("#FFEB9C").setFontColor("#9C6500"); if (d.action==="Full Meeting") actionCell.setBackground("#FFC7CE").setFontColor("#9C0006"); return ContentService.createTextOutput(JSON.stringify({status:"ok"})) .setMimeType(ContentService.MimeType.JSON); } catch(err) { return ContentService.createTextOutput(JSON.stringify({status:"error",message:err.toString()})) .setMimeType(ContentService.MimeType.JSON); } } function doGet() { return ContentService.createTextOutput("Pulse Check API running.") .setMimeType(ContentService.MimeType.TEXT); }
